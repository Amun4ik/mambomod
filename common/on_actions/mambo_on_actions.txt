on_startup = {
    every_province = {
        limit = {
            continent = africa
			has_port = no
			NOT = { terrain = mountains }
			NOT = { region = maghreb_region }
			NOT = { region = egypt_region }
			NOT = { area = south_african_plateau_area }
			NOT = { area = transvaal_area }
			NOT = { area = chokwe_area }
        }
        add_province_triggered_modifier = mambo_malaria
    }
}
on_war_won = {
	ROOT = {
 		add_war_exhaustion = -4
	}
	FROM = {
	add_war_exhaustion = 6
	}
}
on_country_released - {
	THIS = {
		every_owned_province = {
			add_nationalism = -30
		}	
	}
}
